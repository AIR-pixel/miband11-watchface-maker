# 装配用例：整份 .face 的跨语言逐字节比对（Java FaceBuilder ↔ Python 参考 packer）
#
# 格式（| 分隔，5 字段）：
#   name | hasAod | aodCase | nFiles | extraLastBytes
#
# nFiles / extraLastBytes 决定主文件清单（规则见 jvm/DiffMain.java 的 filesFor()，
# Python 侧 tools/diff/reference.py 的 files_for() 必须与之逐字一致）。
# extraLastBytes 专门用来把「文件数据区总长的模 4 余数」走遍 0/1/2/3 ——
# 「尾部只对齐了一半」那个 bug（n=8 看着对、n=12 短 2 字节）就靠这组覆盖。
#
# aodCase 为空时写 "-"，且 hasAod 必须为 0。

# ---- 无 AOD ----
noaod_n1|0|-|1|0
noaod_n3|0|-|3|0
noaod_n8|0|-|8|0
noaod_n12|0|-|12|0
noaod_n12_pad1|0|-|12|1
noaod_n12_pad2|0|-|12|2
noaod_n12_pad3|0|-|12|3
noaod_n24|0|-|24|0
